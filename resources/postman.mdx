## Authentication Setup

### Pre-request Script for OAuth

The collection includes automatic authentication handling:

```javascript
// Pre-request script for automatic token management
const clientId = pm.environment.get("client_id");
const clientSecret = pm.environment.get("client_secret");
const baseUrl = pm.environment.get("base_url");
const currentToken = pm.environment.get("access_token");
const tokenExpiresAt = pm.environment.get("token_expires_at");

// Check if token is expired or missing
function isTokenExpired() {
    if (!currentToken || !tokenExpiresAt) return true;
    const now = Date.now();
    const expiryTime = parseInt(tokenExpiresAt);
    return now >= (expiryTime - 60000); // Refresh 1 minute before expiry
}

// Get new access token
function refreshToken() {
    const tokenRequest = {
        url: `${baseUrl}/oauth/token`,
        method: 'POST',
        header: {
            'Content-Type': 'application/x-www-form-urlencoded'
        },
        body: {
            mode: 'urlencoded',
            urlencoded: [
                {key: 'grant_type', value: 'client_credentials'},
                {key: 'client_id', value: clientId},
                {key: 'client_secret', value: clientSecret},
                {key: 'scope', value: 'accounts:read transactions:read payments:write'}
            ]
        }
    };
    
    pm.sendRequest(tokenRequest, function (err, response) {
        if (err) {
            console.log('Token refresh failed:', err);
            return;
        }
        
        const jsonResponse = response.json();
        if (jsonResponse.access_token) {
            const expiresAt = Date.now() + (jsonResponse.expires_in * 1000);
            pm.environment.set("access_token", jsonResponse.access_token);
            pm.environment.set("token_expires_at", expiresAt.toString());
            console.log('Token refreshed successfully');
        }
    });
}

// Refresh token if needed
if (isTokenExpired()) {
    refreshToken();
}
```

### Test Script for Response Validation

```javascript
// Test script for automatic response validation
pm.test("Status code is successful", function () {
    pm.expect(pm.response.code).to.be.oneOf([200, 201, 202]);
});

pm.test("Response time is acceptable", function () {
    pm.expect(pm.response.responseTime).to.be.below(5000);
});

pm.test("Response has required headers", function () {
    pm.expect(pm.response.headers.get("Content-Type")).to.include("application/json");
    pm.expect(pm.response.headers.get("X-Request-ID")).to.not.be.null;
});

pm.test("Response body is valid JSON", function () {
    pm.response.to.have.jsonBody();
});

// API-specific tests
if (pm.response.code === 200) {
    const jsonData = pm.response.json();
    
    // Test for accounts endpoint
    if (pm.request.url.path.includes('accounts')) {
        pm.test("Accounts response has data structure", function () {
            pm.expect(jsonData).to.have.property('data');
            pm.expect(jsonData.data).to.have.property('accounts');
            pm.expect(jsonData.data.accounts).to.be.an('array');
        });
        
        // Store first account ID for subsequent requests
        if (jsonData.data.accounts.length > 0) {
            pm.environment.set("first_account_id", jsonData.data.accounts[0].accountId);
        }
    }
    
    // Test for payment endpoints
    if (pm.request.url.path.includes('payments')) {
        pm.test("Payment response has status", function () {
            pm.expect(jsonData).to.have.property('data');
            pm.expect(jsonData.data).to.have.property('status');
        });
    }
}
```

## Certificate Configuration

### Setting up mTLS in Postman

<Steps>
  <Step title="Open Postman Settings">
    Go to Settings → Certificates tab
  </Step>
  <Step title="Add Client Certificate">
    Click "Add Certificate" and configure:
    - Host: `*.adcb.com` (or specific domain)
    - CRT file: Your client certificate (.crt or .pem)
    - KEY file: Your private key (.key or .pem)
    - Passphrase: If your key is encrypted
  </Step>
  <Step title="Verify Certificate">
    Test with the "Health Check" request to ensure certificate is working
  </Step>
</Steps>

<Warning>
  Store certificates securely and never commit them to version control. Use Postman's certificate management for team sharing.
</Warning>

## Monitoring and Reports

### Newman CLI Integration

Run collections from command line using Newman:

```bash
# Install Newman
npm install -g newman

# Run collection
newman run "ADCB Open Banking APIs.postman_collection.json" \
  --environment "ADCB Sandbox.postman_environment.json" \
  --reporters html,cli \
  --reporter-html-export report.html

# Run with data file
newman run collection.json \
  --environment environment.json \
  --data test-data.csv \
  --iteration-count 3
```

### CI/CD Integration

**GitHub Actions Example:**
```yaml
name: API Tests
on: [push, pull_request]

jobs:
  api-tests:
    runs-on: ubuntu-latest
    steps:
    - uses: actions/checkout@v2
    
    - name: Install Newman
      run: npm install -g newman
    
    - name: Run Postman Tests
      run: |
        newman run ./postman/collection.json \
          --environment ./postman/sandbox.json \
          --env-var "client_id=${{ secrets.ADCB_CLIENT_ID }}" \
          --env-var "client_secret=${{ secrets.ADCB_CLIENT_SECRET }}" \
          --reporters cli,junit \
          --reporter-junit-export results.xml
    
    - name: Publish Test Results
      uses: actions/upload-artifact@v2
      with:
        name: test-results
        path: results.xml
```

## Troubleshooting

<AccordionGroup>
  <Accordion title="Authentication Issues">
    **Problem**: 401 Unauthorized errors
    
    **Solutions**:
    - Verify client credentials in environment variables
    - Check certificate configuration
    - Ensure token hasn't expired
    - Validate scope requirements
    
    **Debug Steps**:
    ```javascript
    // Add to test script for debugging
    console.log("Current token:", pm.environment.get("access_token"));
    ```

  </Accordion>
  
  <Accordion title="Certificate Problems">
    **Problem**: SSL handshake failures
    
    **Solutions**:
    - Verify certificate format (PEM/P12)
    - Check certificate expiry date
    - Ensure private key matches certificate
    - Validate certificate chain
    
    **Testing Certificate**:
    ```bash
    # Test certificate outside Postman
    ```

    - Verify test assertions match current API responses

    - Review API changes in changelog

    ```javascript
    // More flexible balance check
    ```
