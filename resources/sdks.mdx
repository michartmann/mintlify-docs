## Overview

ADCB provides official SDKs (Software Development Kits) for popular programming languages, making it easy to integrate our Open Banking APIs into your applications. These SDKs handle authentication, request signing, error handling, and other common tasks.

<CardGroup cols={2}>
  <Card title="Easy Integration" icon="plug">
    Pre-built methods for all API endpoints with proper error handling
  </Card>
  <Card title="Built-in Security" icon="shield">

  </Card>
  <Card title="Type Safety" icon="check-circle">
    Full TypeScript definitions and strongly-typed responses

    ```javascript
    const { OpenBankingClient } = require('@adcb/openbanking-sdk');

    const client = new OpenBankingClient({
        clientId: process.env.ADCB_CLIENT_ID,
        clientSecret: process.env.ADCB_CLIENT_SECRET,

            const auth = await client.authenticate({

            if (error.name === 'ADCBAPIError') {

                console.error(`Trace ID: ${error.traceId}`);

                console.error('Unexpected error:', error);
            }
        }
    }
    
    main();
    ```

    ```java
    import com.adcb.openbanking.models.*;
    import java.util.Arrays;
    import java.util.List;
    
    public class ADCBExample {
        public static void main(String[] args) {
            // Initialize client
            OpenBankingClient client = OpenBankingClient.builder()
                .clientId(System.getenv("ADCB_CLIENT_ID"))
                .clientSecret(System.getenv("ADCB_CLIENT_SECRET"))
                .environment(Environment.SANDBOX)
                .certPath("./certs/cert.pem")
                .keyPath("./certs/key.pem")
                .build();
            
            try {
                // Authenticate
                AccessToken token = client.authenticate(
                    Arrays.asList("accounts:read", "transactions:read")
                );
                
                // Get accounts
                List<Account> accounts = client.accounts().list();
                
                for (Account account : accounts) {
                    System.out.println("Account: " + account.getId());
                    
                    // Get balance
                    Balance balance = client.accounts().getBalance(account.getId());
                    System.out.println("Balance: " + balance.getAmount() + " " + balance.getCurrency());
                    
                    // Get transactions
                    TransactionQuery query = TransactionQuery.builder()
                        .fromDate(LocalDate.of(2024, 1, 1))
                        .limit(10)
                        .build();
                        
                    List<Transaction> transactions = client.transactions()
                        .list(account.getId(), query);
                    System.out.println("Recent transactions: " + transactions.size());
                }
                
            } catch (ADCBAPIException e) {
                System.err.println("API Error: " + e.getMessage());
                System.err.println("Status Code: " + e.getStatusCode());
                System.err.println("Trace ID: " + e.getTraceId());
            } catch (Exception e) {
                System.err.println("Unexpected error: " + e.getMessage());
            }
        }
    }
    ```

  <Tab title="Quick Start">
    ```go
    package main
    
    import (
        "context"
        "fmt"
        "log"
        "os"
        "time"
        
        "github.com/adcb/openbanking-go-sdk/adcb"
    )
    
    func main() {
        // Initialize client
        client, err := adcb.NewClient(&adcb.Config{
            ClientID:     os.Getenv("ADCB_CLIENT_ID"),
            ClientSecret: os.Getenv("ADCB_CLIENT_SECRET"),
            Environment:  adcb.Sandbox,
            CertPath:     "./certs/cert.pem",
            KeyPath:      "./certs/key.pem",
            Timeout:      30 * time.Second,
        })
        if err != nil {
            log.Fatal("Failed to create client:", err)
        }
        defer client.Close()
        
        ctx := context.Background()
        
        // Authenticate
        token, err := client.Authenticate(ctx, []string{"accounts:read", "transactions:read"})
        if err != nil {
            log.Fatal("Authentication failed:", err)
        }
        
        // Get accounts
        accounts, err := client.Accounts.List(ctx)
        if err != nil {
            log.Fatal("Failed to get accounts:", err)
        }
        
        for _, account := range accounts {
            fmt.Printf("Account: %s\n", account.ID)
            
            // Get balance
            balance, err := client.Accounts.GetBalance(ctx, account.ID)
            if err != nil {
                log.Printf("Failed to get balance for %s: %v", account.ID, err)
                continue
            }
            fmt.Printf("Balance: %.2f %s\n", balance.Amount, balance.Currency)
            
            // Get transactions
            transactions, err := client.Transactions.List(ctx, account.ID, &adcb.TransactionQuery{
                FromDate: time.Date(2024, 1, 1, 0, 0, 0, 0, time.UTC),
                Limit:    10,
            })
            if err != nil {
                log.Printf("Failed to get transactions for %s: %v", account.ID, err)
                continue
            }
            fmt.Printf("Recent transactions: %d\n", len(transactions))
        }
    }
    ```

    ```go
    // Custom HTTP client with retry
    import (
        "net/http"
        "github.com/hashicorp/go-retryablehttp"
    )
    
    retryClient := retryablehttp.NewClient()
    retryClient.RetryMax = 3
    
    client, err := adcb.NewClient(&adcb.Config{
        ClientID:     clientID,
        ClientSecret: clientSecret,
        HTTPClient:   retryClient.StandardClient(),
    })
    
    // Context with timeout
    ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
    defer cancel()
    
    // Concurrent requests
    type result struct {
        AccountID string
        Balance   *adcb.Balance
        Error     error
    }
    
    results := make(chan result, len(accounts))
    
    for _, account := range accounts {
        go func(accountID string) {
            balance, err := client.Accounts.GetBalance(ctx, accountID)
            results <- result{accountID, balance, err}
        }(account.ID)
    }
    
    for i := 0; i < len(accounts); i++ {
        res := <-results
        if res.Error != nil {
            log.Printf("Error for account %s: %v", res.AccountID, res.Error)
        } else {
            fmt.Printf("Account %s: %.2f %s\n", res.AccountID, res.Balance.Amount, res.Balance.Currency)
        }
    }
    ```

## Common SDK Operations

### Authentication

All SDKs support multiple authentication flows:

<AccordionGroup>
  <Accordion title="Client Credentials Flow">
    Best for server-to-server applications without user interaction.
    
    ```python
    # Client credentials - no user involved
    token = client.authenticate(scopes=['accounts:read'])
    ```
  </Accordion>
  
  <Accordion title="Authorization Code Flow">
    For applications that act on behalf of users.
    
    ```python
    # Step 1: Get authorization URL
    auth_url = client.get_authorization_url(
        scopes=['accounts:read'],
        redirect_uri='https://yourapp.com/callback',
        state='random_state_string'
    )
    
    # Step 2: Exchange code for token
    token = client.exchange_authorization_code(
        code='auth_code_from_callback',
        redirect_uri='https://yourapp.com/callback'
    )
    ```
  </Accordion>
  
  <Accordion title="Token Refresh">
    Automatically refresh expired tokens.
    
    ```python
    # Enable automatic token refresh
    client.auto_refresh_token = True
    
    # Or manually refresh
    new_token = client.refresh_token(refresh_token)
    ```
  </Accordion>
</AccordionGroup>

### Error Handling

All SDKs provide structured error handling:

<CodeGroup>

```python Python
from adcb.exceptions import (
    ADCBAPIError,
    AuthenticationError,
    RateLimitError,
    ValidationError,
    NetworkError
)

try:
    accounts = client.accounts.list()
except AuthenticationError as e:
    # Handle authentication failure
    print(f"Auth error: {e.message}")
    # Refresh token and retry
except RateLimitError as e:
    # Handle rate limiting
    print(f"Rate limited. Retry after: {e.retry_after}")
    time.sleep(e.retry_after)
except ValidationError as e:
    # Handle validation errors
    for error in e.field_errors:
        print(f"Field {error.field}: {error.message}")
except NetworkError as e:
    # Handle network issues
    print(f"Network error: {e.message}")
except ADCBAPIError as e:
    # Handle general API errors
    print(f"API error: {e.message} (Status: {e.status_code})")
```

```javascript JavaScript
try {
    const accounts = await client.accounts.list();
} catch (error) {
    if (error instanceof ADCBAuthenticationError) {
        console.log('Auth error:', error.message);
        // Refresh token and retry
    } else if (error instanceof ADCBRateLimitError) {
        console.log(`Rate limited. Retry after: ${error.retryAfter}s`);
        await sleep(error.retryAfter * 1000);
    } else if (error instanceof ADCBValidationError) {
        console.log('Validation errors:', error.fieldErrors);
    } else if (error instanceof ADCBAPIError) {
        console.log(`API error: ${error.message} (${error.statusCode})`);
        console.log(`Trace ID: ${error.traceId}`);
    }
}
```

</CodeGroup>

### Pagination

Handle paginated responses consistently:

<CodeGroup>

```python Python
# Manual pagination
accounts = client.accounts.list(limit=20)
while accounts.has_next:
    # Process current batch
    for account in accounts:
        print(f"Processing account: {account.id}")
    
    # Get next page
    accounts = accounts.get_next()

# Automatic pagination
for account in client.accounts.list_all():
    print(f"Account: {account.id}")
```

```javascript JavaScript
// Manual pagination
let accounts = await client.accounts.list({ limit: 20 });
while (accounts.hasNext) {
    // Process current batch
    for (const account of accounts.data) {
        console.log(`Processing account: ${account.id}`);
    }
    
    // Get next page
    accounts = await accounts.getNext();
```

## Migration Guides

### Upgrading SDK Versions

<Warning>
  Always review the changelog before upgrading SDK versions. Breaking changes are clearly documented with migration instructions.
</Warning>

<Tabs>
  <Tab title="Python SDK v1.x → v2.x">
    **Major Changes:**
    - Async client is now the default
    - Configuration structure changed
    - Error classes restructured
    
    ```python
    # Before (v1.x)
    from adcb import Client
    client = Client(client_id, client_secret, environment='sandbox')
    accounts = client.get_accounts()
    
    # After (v2.x)
    from adcb import OpenBankingClient
    client = OpenBankingClient(
        client_id=client_id,
        client_secret=client_secret,
        environment='sandbox'
    )
    accounts = client.accounts.list()  # or await client.accounts.list() for async
    ```
  </Tab>
  
  <Tab title="Node.js SDK v1.x → v2.x">
    **Major Changes:**
    - TypeScript rewrite with better types
    - Promise-based API (no more callbacks)
    - New authentication flow
    
    ```javascript
    // Before (v1.x)
    const adcb = require('adcb-sdk');
    const client = new adcb.Client(config);
    client.getAccounts((err, accounts) => {
        // Handle callback
    });
    
    // After (v2.x)
    const { OpenBankingClient } = require('@adcb/openbanking-sdk');
    const client = new OpenBankingClient(config);
    const accounts = await client.accounts.list();
    ```
