```python
import os
from cryptography.fernet import Fernet

class SecureConfig:
    def __init__(self):
        self.encryption_key = os.environ.get('ENCRYPTION_KEY')
        self.cipher = Fernet(self.encryption_key.encode())
    
    def get_credential(self, credential_name: str) -> str:
```

### Caching Strategies

<AccordionGroup>
  <Accordion title="Multi-Layer Caching">
    **Cache Hierarchy:**
    1. **In-Memory Cache** (L1): Fastest access for frequently used data
    2. **Redis Cache** (L2): Shared cache across application instances  
    3. **HTTP Cache** (L3): Browser/proxy caching with proper headers
    
    ```python
    import redis
    from functools import wraps
    import json
    import hashlib
    
    class MultiLayerCache:
        def __init__(self):
            self.memory_cache = {}  # L1 cache
            self.redis_client = redis.Redis(host='localhost', port=6379, db=0)  # L2 cache
        
        def cache_key(self, func_name: str, *args, **kwargs) -> str:
            key_data = f"{func_name}:{args}:{sorted(kwargs.items())}"
            return hashlib.md5(key_data.encode()).hexdigest()
        
        def cached(self, ttl: int = 300):
            def decorator(func):
                @wraps(func)
                def wrapper(*args, **kwargs):
                    cache_key = self.cache_key(func.__name__, *args, **kwargs)
                    
                    # Try L1 cache first
                    if cache_key in self.memory_cache:
                        return self.memory_cache[cache_key]
                    
                    # Try L2 cache
                    cached_value = self.redis_client.get(cache_key)
                    if cached_value:
                        result = json.loads(cached_value)
                        self.memory_cache[cache_key] = result  # Populate L1
                        return result
                    
                    # Execute function and cache result
                    result = func(*args, **kwargs)
                    
                    # Store in both caches
                    self.memory_cache[cache_key] = result
                    self.redis_client.setex(cache_key, ttl, json.dumps(result))
                    
                    return result
                return wrapper
            return decorator
    
    cache = MultiLayerCache()
    
    @cache.cached(ttl=600)  # 10 minutes
    def get_account_balance(account_id: str):
        return adcb_client.get_balance(account_id)
    ```

### Batch Operations

```python
import asyncio
from typing import List, Dict, Any

class BatchProcessor:
    def __init__(self, batch_size: int = 10, max_concurrent: int = 5):
        self.batch_size = batch_size
        self.semaphore = asyncio.Semaphore(max_concurrent)
    
    async def process_accounts_batch(self, account_ids: List[str]) -> Dict[str, Any]:
        """Process multiple accounts efficiently"""
        results = {}
        
        # Split into batches
        batches = [account_ids[i:i + self.batch_size] 
                  for i in range(0, len(account_ids), self.batch_size)]
        
        # Process batches concurrently
        batch_tasks = [self._process_single_batch(batch) for batch in batches]
        batch_results = await asyncio.gather(*batch_tasks, return_exceptions=True)
        
        # Combine results
        for batch_result in batch_results:
            if isinstance(batch_result, dict):
                results.update(batch_result)
            else:
                logging.error(f"Batch processing error: {batch_result}")
        
        return results
    
    async def _process_single_batch(self, account_ids: List[str]) -> Dict[str, Any]:
        async with self.semaphore:
            tasks = []
            for account_id in account_ids:
                task = self._get_account_with_balance(account_id)
                tasks.append(task)
            
            results = await asyncio.gather(*tasks, return_exceptions=True)
            return {aid: result for aid, result in zip(account_ids, results) 
                   if not isinstance(result, Exception)}
    
    async def _get_account_with_balance(self, account_id: str):
        """Get account details and balance in parallel"""
        account_task = self.client.get_account(account_id)
        balance_task = self.client.get_balance(account_id)
        
        account, balance = await asyncio.gather(account_task, balance_task)
        return {'account': account, 'balance': balance}
```

## Reliability Patterns

### Circuit Breaker Implementation

<CodeGroup>

```python Python
import time
import threading
from enum import Enum
from typing import Callable, Any, Optional

class CircuitState(Enum):
    CLOSED = "closed"
    OPEN = "open"  
    HALF_OPEN = "half_open"

class CircuitBreaker:
    def __init__(
        self,
        failure_threshold: int = 5,
        recovery_timeout: int = 60,
        expected_exception: type = Exception,
        success_threshold: int = 3  # For half-open state
    ):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.expected_exception = expected_exception
        self.success_threshold = success_threshold
        
        self.failure_count = 0
        self.success_count = 0
        self.last_failure_time: Optional[float] = None
        self.state = CircuitState.CLOSED
        self._lock = threading.RLock()
        
        # Metrics
        self.total_requests = 0
        self.total_failures = 0
    
    def call(self, func: Callable[..., Any], *args, **kwargs) -> Any:
        with self._lock:
            self.total_requests += 1
            
            if self.state == CircuitState.OPEN:
                if self._should_attempt_reset():
                    self.state = CircuitState.HALF_OPEN
                    self.success_count = 0
                else:
                    raise Exception(f"Circuit breaker is OPEN. Last failure: {self.last_failure_time}")
            
            try:
                result = func(*args, **kwargs)
                self._on_success()
                return result
                
            except self.expected_exception as e:
                self._on_failure()
                raise
    
    def _should_attempt_reset(self) -> bool:
        return (
            self.last_failure_time is not None and
            time.time() - self.last_failure_time >= self.recovery_timeout
        )
    
    def _on_success(self):
        if self.state == CircuitState.HALF_OPEN:
            self.success_count += 1
            if self.success_count >= self.success_threshold:
                self.state = CircuitState.CLOSED
                self.failure_count = 0
        else:
            self.failure_count = 0
    
    def _on_failure(self):
        self.total_failures += 1
        self.failure_count += 1
        self.last_failure_time = time.time()
        
        if self.failure_count >= self.failure_threshold:
            self.state = CircuitState.OPEN
    
    @property
    def failure_rate(self) -> float:
        if self.total_requests == 0:
            return 0.0
        return self.total_failures / self.total_requests
    
    def reset(self):
        """Manually reset circuit breaker"""
        with self._lock:
            self.state = CircuitState.CLOSED
            self.failure_count = 0
            self.success_count = 0
            self.last_failure_time = None

# Usage example
payment_circuit = CircuitBreaker(
    failure_threshold=3,
    recovery_timeout=30,
    expected_exception=ADCBAPIError
)

def make_payment(payment_data):
    return payment_circuit.call(adcb_client.create_payment, payment_data)
```

### Bulkhead Pattern

```python
import asyncio
import threading
from concurrent.futures import ThreadPoolExecutor
from typing import Dict, Any

class BulkheadManager:
    """Isolate different types of operations to prevent cascading failures"""
    
    def __init__(self):
        # Separate thread pools for different operations
        self.pools = {
            'account_ops': ThreadPoolExecutor(max_workers=10, thread_name_prefix='account'),
            'payment_ops': ThreadPoolExecutor(max_workers=5, thread_name_prefix='payment'),  
            'webhook_ops': ThreadPoolExecutor(max_workers=3, thread_name_prefix='webhook'),
            'analytics_ops': ThreadPoolExecutor(max_workers=15, thread_name_prefix='analytics')
        }
        
        # Separate semaphores for async operations
        self.semaphores = {
            'account_ops': asyncio.Semaphore(20),
            'payment_ops': asyncio.Semaphore(10),
            'webhook_ops': asyncio.Semaphore(5),
            'analytics_ops': asyncio.Semaphore(30)
        }
    
    async def execute_account_operation(self, func, *args, **kwargs):
        async with self.semaphores['account_ops']:
            loop = asyncio.get_event_loop()
            return await loop.run_in_executor(
                self.pools['account_ops'], 
                func, *args, **kwargs
            )
    
    async def execute_payment_operation(self, func, *args, **kwargs):
        async with self.semaphores['payment_ops']:
            loop = asyncio.get_event_loop()
            return await loop.run_in_executor(
                self.pools['payment_ops'],
                func, *args, **kwargs
            )
    
    def shutdown(self):
        """Gracefully shutdown all thread pools"""
        for pool in self.pools.values():
            pool.shutdown(wait=True)

# Usage
bulkhead = BulkheadManager()

# Account operations won't block payment operations
account_task = bulkhead.execute_account_operation(get_account_balance, account_id)
payment_task = bulkhead.execute_payment_operation(create_payment, payment_data)

results = await asyncio.gather(account_task, payment_task)
```

```python
class GracefulDegradationManager:
    """Handle service degradation gracefully"""
    
    def __init__(self):
        self.service_health = {
            'accounts': True,
            'payments': True,
            'webhooks': True,
            'analytics': True
        }
        self.fallback_data = {}
    
    def check_service_health(self, service: str) -> bool:
        """Check if service is healthy"""
        try:
            response = requests.get(f'{base_url}/health/{service}', timeout=5)
            healthy = response.status_code == 200
            self.service_health[service] = healthy
            return healthy
        except:
            self.service_health[service] = False
            return False
    
    def get_account_balance_with_fallback(self, account_id: str):
        """Get balance with fallback to cached data"""
        if self.check_service_health('accounts'):
            try:
                return adcb_client.get_balance(account_id)
            except Exception as e:
                logging.warning(f"Primary balance service failed: {e}")
        
        # Fallback to cached balance
        cached_balance = self.get_cached_balance(account_id)
        if cached_balance:
            cached_balance['warning'] = 'Using cached data - real-time balance unavailable'
            return cached_balance
        
        # Final fallback
        return {
            'account_id': account_id,
            'balance': 'unavailable',
            'error': 'Balance service temporarily unavailable'
        }
    
    def get_cached_balance(self, account_id: str):
        """Retrieve last known balance from cache"""
        return self.fallback_data.get(f'balance:{account_id}')
    
    def update_fallback_data(self, key: str, data: dict):
        """Update fallback data cache"""
        self.fallback_data[key] = {
            **data,
            'cached_at': datetime.utcnow().isoformat()
        }
```

### Application Metrics

<CodeGroup>

```python Python
import time
import threading
from collections import defaultdict, deque
from typing import Dict, List
import json

class MetricsCollector:
    def __init__(self, window_size: int = 300):  # 5-minute windows
        self.window_size = window_size
        self.metrics = defaultdict(lambda: defaultdict(deque))
        self.lock = threading.RLock()
    
    def record_api_call(
        self,
        endpoint: str,
        method: str,
        status_code: int,
        duration: float,
        error_type: str = None
    ):
        """Record API call metrics"""
        timestamp = time.time()
        
        with self.lock:
            key = f"{method}:{endpoint}"
            
            # Record response time
            self.metrics[key]['response_times'].append((timestamp, duration))
            
            # Record status codes
            self.metrics[key]['status_codes'].append((timestamp, status_code))
            
            # Record errors
            if error_type:
                self.metrics[key]['errors'].append((timestamp, error_type))
            
            # Clean old data
            self._cleanup_old_data(key, timestamp)
    
    def get_metrics_summary(self) -> Dict:
        """Get current metrics summary"""
        summary = {}
        current_time = time.time()
        
        with self.lock:
            for endpoint, data in self.metrics.items():
                # Calculate response time stats
                response_times = [duration for _, duration in data['response_times']]
                if response_times:
                    avg_response_time = sum(response_times) / len(response_times)
                    p95_response_time = sorted(response_times)[int(len(response_times) * 0.95)]
                else:
                    avg_response_time = p95_response_time = 0
                
                # Calculate error rate
                total_requests = len(data['status_codes'])
                error_requests = len([1 for _, code in data['status_codes'] if code >= 400])
                error_rate = error_requests / total_requests if total_requests > 0 else 0
                
                summary[endpoint] = {
                    'total_requests': total_requests,
                    'error_rate': error_rate,
                    'avg_response_time': avg_response_time,
                    'p95_response_time': p95_response_time,
                    'requests_per_minute': len([1 for ts, _ in data['status_codes'] 
                                              if current_time - ts <= 60])
                }
        
        return summary
    
    def _cleanup_old_data(self, key: str, current_time: float):
        """Remove data outside the time window"""
        cutoff_time = current_time - self.window_size
        
        for metric_type in ['response_times', 'status_codes', 'errors']:
            metric_data = self.metrics[key][metric_type]
            while metric_data and metric_data[0][0] < cutoff_time:
                metric_data.popleft()

# Decorator for automatic metrics collection
def monitor_api_call(endpoint: str, method: str):
    def decorator(func):
        def wrapper(*args, **kwargs):
            start_time = time.time()
            error_type = None
            status_code = 200
            
            try:
                result = func(*args, **kwargs)
                if hasattr(result, 'status_code'):
                    status_code = result.status_code
                return result
            except Exception as e:
                error_type = type(e).__name__
                if hasattr(e, 'status_code'):
                    status_code = e.status_code
                else:
                    status_code = 500
                raise
            finally:
                duration = time.time() - start_time
                metrics_collector.record_api_call(
                    endpoint, method, status_code, duration, error_type
                )
        
        return wrapper
    return decorator

# Usage
```

```python
from dataclasses import dataclass
from typing import Dict, List, Optional
import requests

@dataclass
class HealthCheckResult:
    service: str
    status: str  # 'healthy', 'degraded', 'unhealthy'
    response_time: float
    details: Optional[Dict] = None
    error: Optional[str] = None

class HealthChecker:
    def __init__(self):
        self.checks = {
            'adcb_api': self._check_adcb_api,
            'database': self._check_database,
            'redis': self._check_redis,
            'external_services': self._check_external_services
        }
    
    def run_all_checks(self) -> Dict[str, HealthCheckResult]:
        """Run all health checks"""
        results = {}
        
        for check_name, check_func in self.checks.items():
            try:
                results[check_name] = check_func()
            except Exception as e:
                results[check_name] = HealthCheckResult(
                    service=check_name,
                    status='unhealthy',
                    response_time=0,
                    error=str(e)
                )
        
        return results
    
    def _check_adcb_api(self) -> HealthCheckResult:
        """Check ADCB API connectivity"""
        start_time = time.time()
        
        try:
            response = requests.get(
                f'{base_url}/health',
                headers={'Authorization': f'Bearer {access_token}'},
                timeout=5
            )
            response_time = time.time() - start_time
            
            if response.status_code == 200:
                return HealthCheckResult(
                    service='adcb_api',
                    status='healthy',
                    response_time=response_time,
                    details={'endpoint': f'{base_url}/health'}
                )
            else:
                return HealthCheckResult(
                    service='adcb_api',
                    status='degraded',
                    response_time=response_time,
                    details={'status_code': response.status_code}
                )
        
        except requests.exceptions.Timeout:
            return HealthCheckResult(
                service='adcb_api',
                status='unhealthy',
                response_time=5.0,
                error='Timeout after 5 seconds'
            )
        except Exception as e:
            return HealthCheckResult(
                service='adcb_api',
                status='unhealthy',
                response_time=0,
                error=str(e)
            )
    
    def get_overall_health(self, results: Dict[str, HealthCheckResult]) -> str:
        """Determine overall application health"""
        unhealthy_count = sum(1 for r in results.values() if r.status == 'unhealthy')
        degraded_count = sum(1 for r in results.values() if r.status == 'degraded')
        
        if unhealthy_count > 0:
            return 'unhealthy'
        elif degraded_count > 0:
            return 'degraded'
        else:
            return 'healthy'

    }
    
    status_code = 200 if overall_health == 'healthy' else 503
    return jsonify(response_data), status_code
```

    **Incident Severity Levels:**
    - **P0**: Complete API outage affecting customers
    - **P1**: Partial functionality down, significant impact
    - **P2**: Degraded performance, workarounds available  
    - **P3**: Minor issues, no customer impact
    
    **Response Procedures:**
    1. **Immediate Response** (5 minutes)
       - Acknowledge incident
       - Check API status page
       - Verify monitoring systems
    
    2. **Investigation** (15 minutes)
       - Review recent deployments
       - Check error logs and metrics
       - Validate certificate expiry
    
    3. **Resolution** (varies)
       - Apply fix or rollback
       - Verify fix in staging first
       - Monitor for improvement
    
    4. **Post-Incident** (24 hours)
       - Write incident report
       - Identify root cause
       - Implement prevention measures
