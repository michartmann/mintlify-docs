## Overview

ADCB's sandbox environment provides a safe, production-like testing environment where you can develop and test your integration without affecting real customer data or incurring transaction costs. The sandbox mimics all production functionality while using test data.

<Info>
  The sandbox environment is completely isolated from production and uses synthetic test data. No real customer information or actual transactions are processed.
</Info>

import requests
import pytest
from datetime import datetime, timedelta

class TestADCBSandbox:
    def __init__(self):
        self.base_url = "https://sandbox-api.adcb.com"
        self.access_token = None
    
    def authenticate(self):
        """Test OAuth authentication"""
        response = requests.post(
            f"{self.base_url}/oauth/token",
            data={
                "grant_type": "client_credentials",
                "client_id": "sandbox_client_12345",
                "client_secret": "sandbox_secret_67890",
                "scope": "accounts:read transactions:read"
            },
            cert=("./certs/sandbox-cert.pem", "./certs/sandbox-key.pem")
        )
        
        assert response.status_code == 200, f"Auth failed: {response.text}"
        self.access_token = response.json()["access_token"]
        return self.access_token
    
    def test_get_accounts(self):
        """Test account listing"""
        if not self.access_token:
            self.authenticate()
        
        response = requests.get(
            f"{self.base_url}/v1/accounts",
            headers={"Authorization": f"Bearer {self.access_token}"},
            cert=("./certs/sandbox-cert.pem", "./certs/sandbox-key.pem")
        )
        
        assert response.status_code == 200
        data = response.json()
        assert "data" in data
        assert "accounts" in data["data"]
        assert len(data["data"]["accounts"]) > 0
        
        # Verify account structure
        account = data["data"]["accounts"][0]
        required_fields = ["accountId", "currency", "accountType", "account"]
        for field in required_fields:
            assert field in account, f"Missing field: {field}"
    
    def test_get_account_balance(self):
        """Test balance retrieval"""
        test_account = "AE07033TEST000000000001"
        
        response = requests.get(
            f"{self.base_url}/v1/accounts/{test_account}/balances",
            headers={"Authorization": f"Bearer {self.access_token}"},
            cert=("./certs/sandbox-cert.pem", "./certs/sandbox-key.pem")
        )
        
        assert response.status_code == 200
        data = response.json()
        
        # Verify balance structure
        balance = data["data"]["balance"][0]
        assert "amount" in balance
        assert "currency" in balance
        assert balance["currency"] == "AED"

            f"{self.base_url}/v1/accounts/{test_account}/transactions",
            headers={"Authorization": f"Bearer {self.access_token}"},
            params={
                "fromBookingDateTime": from_date,
                "limit": 10
            },
            cert=("./certs/sandbox-cert.pem", "./certs/sandbox-key.pem")
        )
        
        assert response.status_code == 200
        data = response.json()
        
        # Verify transaction structure
        if data["data"]["transactions"]:
            transaction = data["data"]["transactions"][0]
            required_fields = ["transactionId", "amount", "bookingDate", "status"]
            for field in required_fields:
                assert field in transaction

# Run tests

import asyncio
import aiohttp
import time
from concurrent.futures import ThreadPoolExecutor

class SandboxLoadTester:
    def __init__(self, base_url="https://sandbox-api.adcb.com"):
        self.base_url = base_url
        self.access_token = None
        self.test_accounts = [
            "AE07033TEST000000000001",
            "AE07033TEST000000000002",
            "AE07033TEST000000000003"
        ]
    
    async def authenticate(self):
        """Get access token for load testing"""
        # Implementation similar to previous examples
        pass
    
    async def load_test_accounts(self, concurrent_requests=10, total_requests=100):
        """Load test account endpoints"""
        
        async def make_request(session, semaphore, account_id):
            async with semaphore:
                start_time = time.time()
                try:
                    async with session.get(
                        f"{self.base_url}/v1/accounts/{account_id}/balance",
                        headers={"Authorization": f"Bearer {self.access_token}"}
                    ) as response:
                        status = response.status
                        duration = time.time() - start_time
                        return {"status": status, "duration": duration, "success": status == 200}
                except Exception as e:
                    duration = time.time() - start_time
                    return {"status": 0, "duration": duration, "success": False, "error": str(e)}
        
        # Create semaphore to limit concurrent requests
        semaphore = asyncio.Semaphore(concurrent_requests)
        
        async with aiohttp.ClientSession() as session:
            tasks = []
            for i in range(total_requests):
                account_id = self.test_accounts[i % len(self.test_accounts)]
                task = make_request(session, semaphore, account_id)
                tasks.append(task)
            
            results = await asyncio.gather(*tasks)
        
        # Analyze results
        successful_requests = sum(1 for r in results if r["success"])
        avg_duration = sum(r["duration"] for r in results) / len(results)
        max_duration = max(r["duration"] for r in results)
        min_duration = min(r["duration"] for r in results)
        
        print(f"Load Test Results:")
        print(f"Total Requests: {total_requests}")
        print(f"Successful: {successful_requests} ({successful_requests/total_requests*100:.1f}%)")
        print(f"Average Duration: {avg_duration:.3f}s")
        print(f"Min/Max Duration: {min_duration:.3f}s / {max_duration:.3f}s")
        
        return results

# Usage
async def run_load_test():
    tester = SandboxLoadTester()
    await tester.authenticate()
    results = await tester.load_test_accounts(concurrent_requests=5, total_requests=50)

if __name__ == "__main__":
    asyncio.run(run_load_test())

    **Solutions**:

    # Verify certificate format

    # Test certificate with curl
    curl -v --cert sandbox-cert.pem --key sandbox-key.pem \

  </Accordion>
  
  <Accordion title="Rate Limiting in Sandbox">
    **Issue**: Hitting rate limits during testing
    
    **Solutions**:
    - Sandbox has relaxed limits but they still exist
    - Implement delays between requests in tests
    - Use different test accounts to distribute load
    - Contact support for temporary limit increases
  </Accordion>

    **Issue**: Test data not behaving as expected
    
    **Solutions**:
    - Sandbox data is reset periodically
    - Don't rely on specific balance amounts
    - Use relative comparisons (balance changes)
    - Check sandbox maintenance schedule
  </Accordion>

### Test Reporting

```python
import json
import pytest
from datetime import datetime

class SandboxTestReporter:
    def __init__(self):
        self.results = {
            "timestamp": datetime.utcnow().isoformat(),
            "environment": "sandbox",
            "test_results": []
        }
    
    def add_test_result(self, test_name: str, status: str, duration: float, details: dict = None):
        """Add test result to report"""
        self.results["test_results"].append({
            "test_name": test_name,
            "status": status,  # PASS, FAIL, SKIP
            "duration_seconds": duration,
            "details": details or {}
        })
    
    def generate_report(self, output_file: str = "sandbox-test-report.json"):
        """Generate test report"""
        summary = {
            "total_tests": len(self.results["test_results"]),
            "passed": sum(1 for r in self.results["test_results"] if r["status"] == "PASS"),
            "failed": sum(1 for r in self.results["test_results"] if r["status"] == "FAIL"),
            "skipped": sum(1 for r in self.results["test_results"] if r["status"] == "SKIP"),
            "total_duration": sum(r["duration_seconds"] for r in self.results["test_results"])
        }
        
        self.results["summary"] = summary
        
        with open(output_file, 'w') as f:
            json.dump(self.results, f, indent=2)
        
        print(f"Test Report Generated: {output_file}")
        print(f"Tests: {summary['total_tests']}, "
              f"Passed: {summary['passed']}, "
              f"Failed: {summary['failed']}, "
              f"Duration: {summary['total_duration']:.2f}s")
```
