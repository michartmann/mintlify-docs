    ```python
            }
            
            response = requests.get(
                f"{self.base_url}/accounts/{account_id}/balances",
                headers=headers,
                cert=self.cert
            )
    ```

    ```javascript
    const https = require('https');

    const axios = require('axios');
    
    class ADCBClient {
        constructor(clientId, clientSecret, certPath, keyPath) {
            this.clientId = clientId;
            this.clientSecret = clientSecret;
            this.httpsAgent = new https.Agent({
                cert: fs.readFileSync(certPath),
                key: fs.readFileSync(keyPath),
            });
            this.baseURL = 'https://api.adcb.com/v1';
            this.token = null;
        }
        
        async getToken() {
            try {
                const response = await axios.post(
                    'https://auth.adcb.com/oauth/token',
                    {
                        grant_type: 'client_credentials',
                        client_id: this.clientId,
                        client_secret: this.clientSecret,
                        scope: 'accounts:read'
                    },
                    { httpsAgent: this.httpsAgent }
                );
                
                this.token = response.data.access_token;
                return this.token;
            } catch (error) {
                throw new Error(`Token acquisition failed: ${error.message}`);
            }
        }
        
        async getAccountBalance(accountId) {
            if (!this.token) await this.getToken();
            
            const headers = {
                'Authorization': `Bearer ${this.token}`,
                'X-Request-ID': `req_${Date.now()}`,
                'Content-Type': 'application/json'
            };
            
            try {
                const response = await axios.get(
                    `${this.baseURL}/accounts/${accountId}/balances`,
                    { 
                        headers,
                        httpsAgent: this.httpsAgent 
                    }
                );
                
                return response.data;
            } catch (error) {
                throw new Error(`Balance retrieval failed: ${error.message}`);
            }
        }
    }
    
    // Usage
    ```

    ```java
    import java.net.http.*;
    import java.net.URI;
    import java.security.*;
    import javax.net.ssl.*;
    import java.io.*;
    import com.fasterxml.jackson.databind.ObjectMapper;
    
    public class ADCBClient {
        private final String clientId;
        private final String clientSecret;
        private final HttpClient httpClient;
        private final ObjectMapper mapper;
        private String accessToken;
        
        public ADCBClient(String clientId, String clientSecret, 
                         String certPath, String keyPath) throws Exception {
            this.clientId = clientId;
            this.clientSecret = clientSecret;
            this.mapper = new ObjectMapper();
            
            // Setup mTLS
            KeyStore keyStore = KeyStore.getInstance("PKCS12");
            keyStore.load(new FileInputStream(certPath), "password".toCharArray());
            
            KeyManagerFactory kmf = KeyManagerFactory.getInstance("SunX509");
            kmf.init(keyStore, "password".toCharArray());
            
            SSLContext sslContext = SSLContext.getInstance("TLS");
            sslContext.init(kmf.getKeyManagers(), null, null);
            
            this.httpClient = HttpClient.newBuilder()
                .sslContext(sslContext)
                .build();
        }
        
        public String getAccessToken() throws Exception {
            if (accessToken != null) return accessToken;
            
            String formData = String.format(
                "grant_type=client_credentials&client_id=%s&client_secret=%s&scope=accounts:read",
                clientId, clientSecret
            );
            
            HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create("https://auth.adcb.com/oauth/token"))
                .header("Content-Type", "application/x-www-form-urlencoded")
                .POST(HttpRequest.BodyPublishers.ofString(formData))
                .build();
            
            HttpResponse<String> response = httpClient.send(request, 
                HttpResponse.BodyHandlers.ofString());
            
            TokenResponse tokenResponse = mapper.readValue(response.body(), 
                TokenResponse.class);
            this.accessToken = tokenResponse.getAccessToken();
            
            return accessToken;
        }
        
        public AccountBalance getAccountBalance(String accountId) throws Exception {
            String token = getAccessToken();
            
            HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create("https://api.adcb.com/v1/accounts/" + accountId + "/balances"))
                .header("Authorization", "Bearer " + token)
                .header("X-Request-ID", "req_" + System.currentTimeMillis())
                .header("Content-Type", "application/json")
                .GET()
                .build();
            
            HttpResponse<String> response = httpClient.send(request,
                HttpResponse.BodyHandlers.ofString());
            
            return mapper.readValue(response.body(), AccountBalance.class);
        }
    }
    
    // Usage
    ADCBClient client = new ADCBClient("your_client_id", "your_secret", 
                                      "cert.p12", "key.p12");
    AccountBalance balance = client.getAccountBalance("ACC123456789");
    System.out.println("Available balance: " + balance.getData().getAvailable().getAmount() + " AED");
    ```

### 3. Error Handling & Retry Logic

<Tabs>
  <Tab title="Python">
    ```python
    import time
    import random
    from functools import wraps
    
    def retry_with_backoff(max_retries=3, base_delay=1, max_delay=60):
        """Decorator for exponential backoff retry logic"""
        def decorator(func):
            @wraps(func)
            def wrapper(*args, **kwargs):
                for attempt in range(max_retries + 1):
                    try:
                        return func(*args, **kwargs)
                    except requests.exceptions.RequestException as e:
                        if attempt == max_retries:
                            raise
                        
                        # Handle specific error types
                        if e.response.status_code in [429, 503, 502, 504]:
                            delay = min(base_delay * (2 ** attempt) + random.uniform(0, 1), max_delay)
                            print(f"Retry attempt {attempt + 1} after {delay:.2f}s")
                            time.sleep(delay)
                        else:
                            raise  # Don't retry for 4xx errors (except 429)
                            
                return None
            return wrapper
        return decorator
    
    class ADCBClientWithRetry(ADCBClient):
        @retry_with_backoff(max_retries=3)
        def _make_request(self, method, endpoint, **kwargs):
            """Make HTTP request with retry logic"""
            try:
                response = requests.request(
                    method, 
                    f"{self.base_url}{endpoint}",
                    **kwargs
                )
                
                # Handle API-specific errors
                if response.status_code == 401:
                    self.token = None  # Force token refresh
                    raise requests.exceptions.RequestException("Unauthorized - token expired")
                
                response.raise_for_status()
                return response.json()
                
            except requests.exceptions.Timeout:
                print("Request timeout - will retry")
                raise
            except requests.exceptions.ConnectionError:
                print("Connection error - will retry")
                raise
    ```

```python
class PaymentProcessor:
    def process_payment(self, payment_request):
        """Complete payment processing workflow"""
        
        try:
            # Validate request
            self.validate_payment_request(payment_request)
            
            # Check account balance
            balance = self.check_sufficient_funds(payment_request)

            # Monitor payment status
            status = self.monitor_payment_status(payment["paymentId"])

                "payment_id": payment["paymentId"],

            }
```
