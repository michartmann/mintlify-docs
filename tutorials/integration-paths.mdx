    # Complete OAuth 2.0 implementation
    class AuthManager:
        def __init__(self, client_id, client_secret):
            self.client_id = client_id
            self.client_secret = client_secret
            self.redirect_uri = "http://localhost:8000/callback"
        
        def get_authorization_url(self, state=None):
            """Generate OAuth authorization URL"""
            params = {
                "response_type": "code",
                "client_id": self.client_id,
                "redirect_uri": self.redirect_uri,
                "scope": "accounts:read transactions:read",
                "state": state or self.generate_state()
            }
            
            return f"https://auth.adcb.com/authorize?{urlencode(params)}"
        
        def exchange_code_for_token(self, authorization_code):
            """Exchange authorization code for access token"""
            data = {
                "grant_type": "authorization_code",
                "client_id": self.client_id,
                "client_secret": self.client_secret,
                "code": authorization_code,
                "redirect_uri": self.redirect_uri
            }
            
            response = requests.post(
                "https://auth.adcb.com/oauth/token",
                data=data,
                cert=self.cert_path
            )
            
            return response.json()

    // Token refresh and management
    class TokenManager {
        constructor(clientId, clientSecret) {
            this.clientId = clientId;
            this.clientSecret = clientSecret;
            this.accessToken = null;
            this.refreshToken = null;
            this.expiresAt = null;
        }
        
        async getValidToken() {
            if (this.isTokenValid()) {
                return this.accessToken;
            }
            
            if (this.refreshToken) {
                return await this.refreshAccessToken();
            }
            
            throw new Error('No valid token available. Please re-authenticate.');
        }
        
        async refreshAccessToken() {
            const response = await fetch('https://auth.adcb.com/oauth/token', {
                method: 'POST',
                headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
                body: new URLSearchParams({
                    grant_type: 'refresh_token',
                    refresh_token: this.refreshToken,
                    client_id: this.clientId,
                    client_secret: this.clientSecret
                })
            });
            
            const tokens = await response.json();
            this.updateTokens(tokens);
            
            return this.accessToken;
        }
    }

class BankingDashboard:

<Tabs>
  <Tab title="Category Detection">
    ```python
    class TransactionAnalyzer:
        def __init__(self):
            self.category_rules = {
                'grocery': ['CARREFOUR', 'LULU', 'SPINNEYS', 'WAITROSE'],
                'fuel': ['ENOC', 'EPPCO', 'ADNOC'],
                'dining': ['RESTAURANT', 'CAFE', 'PIZZA', 'MCDONALD'],
                'utilities': ['DEWA', 'SEWA', 'ETISALAT', 'DU'],
                'transport': ['UBER', 'CAREEM', 'METRO', 'TAXI']
            }
        
        def categorize_transaction(self, transaction):
            """Categorize transaction based on merchant name"""
            merchant = transaction.get('merchantName', '').upper()
            
            for category, keywords in self.category_rules.items():
                if any(keyword in merchant for keyword in keywords):
                    return {
                        'category': category,
                        'confidence': 0.9,
                        'subcategory': self.get_subcategory(merchant, category)
                    }
            
            return {
                'category': 'other',
                'confidence': 0.5,
                'subcategory': None
            }
        
        def generate_spending_insights(self, transactions):
            """Generate spending insights from transaction data"""
            categories = {}
            
            for txn in transactions:
                category = self.categorize_transaction(txn)['category']
                amount = float(txn['amount']['amount'])
                
                if category not in categories:
                    categories[category] = {'total': 0, 'count': 0, 'transactions': []}
                
                categories[category]['total'] += amount
                categories[category]['count'] += 1
                categories[category]['transactions'].append(txn)
            
            return {
                'by_category': categories,
                'top_spending_category': max(categories.items(), key=lambda x: x[1]['total']),
                'average_transaction': sum(float(t['amount']['amount']) for t in transactions) / len(transactions)
            }
    ```

                'percentage_used': percentage_used,

            'bankAccount': merchant_data['settlement_account'],

        # Step 4: Generate API credentials

            merchant_account['merchantId']

    class PaymentProcessor:
        def __init__(self):
            self.fraud_detector = FraudDetector()
            self.rate_limiter = RateLimiter()
        
        async def process_payment(self, payment_request):
            """Process payment with full validation and fraud checks"""
            
            # Rate limiting
            if not self.rate_limiter.allow_request(payment_request['merchant_id']):
                raise PaymentError('Rate limit exceeded', code='RATE_LIMIT')
            
            # Fraud detection
            fraud_score = await self.fraud_detector.analyze(payment_request)
            if fraud_score > 0.7:
                return await self.handle_suspicious_payment(payment_request, fraud_score)
            
            # Validate payment details
            validation = await self.validate_payment_request(payment_request)
            if not validation['valid']:
                raise PaymentError(validation['error'], code='VALIDATION_ERROR')
            
            # Create payment consent
            consent = await self.create_payment_consent(payment_request)
            
            # Initiate payment
            payment = await self.initiate_payment(consent['consent_id'])
            
            # Track payment status
            self.track_payment_async(payment['payment_id'])
            
            return {
                'payment_id': payment['payment_id'],
                'status': payment['status'],
                'consent_url': consent['authorization_url'],
                'expires_at': consent['expires_at']
            }

    class CurrencyManager:
        def __init__(self):
            self.exchange_rates = ExchangeRateProvider()
        
        async def process_cross_currency_payment(self, payment_request):
            """Handle payments involving currency conversion"""
            
            from_currency = payment_request['from_account']['currency']
            to_currency = payment_request['to_account']['currency']
            amount = payment_request['amount']
            
            if from_currency == to_currency:
                return await self.process_same_currency_payment(payment_request)
            
            # Get real-time exchange rate
            rate = await self.exchange_rates.get_rate(from_currency, to_currency)
            
            # Calculate conversion with spread
            conversion_rate = rate * (1 - self.get_fx_spread(amount))
            converted_amount = amount * conversion_rate
            
            # Create payment with FX details
            payment_data = {
                **payment_request,
                'fx_details': {
                    'original_amount': amount,
                    'original_currency': from_currency,
                    'converted_amount': converted_amount,
                    'target_currency': to_currency,
                    'exchange_rate': conversion_rate,
                    'fx_fee': amount * self.get_fx_spread(amount)
                }
            }
            
            return await self.process_fx_payment(payment_data)

### Module 3: Settlement & Reconciliation (90 minutes)

Build automated settlement processing:

```python
class SettlementEngine:
    def __init__(self):
        self.scheduler = BackgroundScheduler()
        self.scheduler.start()
        
        # Schedule daily settlement at 2 AM
        self.scheduler.add_job(
            self.process_daily_settlements,
            'cron',
            hour=2,
            minute=0
        )
    
    async def process_daily_settlements(self):
        """Process all pending settlements"""
        
        # Get all merchants due for settlement
        merchants = await self.get_settlement_eligible_merchants()
        
        for merchant in merchants:
            try:
                await self.settle_merchant_transactions(merchant)
            except Exception as e:
                logger.error(f"Settlement failed for merchant {merchant['id']}: {e}")
                await self.handle_settlement_failure(merchant, e)
    
    async def settle_merchant_transactions(self, merchant):
        """Calculate and execute settlement for a merchant"""
        
        # Get all successful transactions since last settlement
        transactions = await self.get_unsettled_transactions(merchant['id'])
        
        # Calculate settlement amount
        gross_amount = sum(t['amount'] for t in transactions)
        fees = self.calculate_merchant_fees(transactions)
        net_settlement = gross_amount - fees
        
        # Execute settlement transfer
        settlement = await self.adcb_client.create_settlement_transfer({
            'merchant_id': merchant['id'],
            'settlement_account': merchant['settlement_account'],
            'amount': net_settlement,
            'currency': 'AED',
            'transactions': [t['id'] for t in transactions],
            'fee_breakdown': fees
        })
        
        # Generate settlement report
        await self.generate_settlement_report(merchant, settlement, transactions)
        
        return settlement
```

### Module 1: Bulk Operations & Data Synchronization

```python
class EnterpriseBulkProcessor:
    def __init__(self):
        self.batch_size = 1000
        self.parallel_workers = 5
    
    async def bulk_account_sync(self, customer_ids):
        """Efficiently sync thousands of accounts"""
        
        # Process in batches to respect rate limits
        batches = [customer_ids[i:i+self.batch_size] 
                  for i in range(0, len(customer_ids), self.batch_size)]
        
        results = []
        
        # Process batches in parallel
        semaphore = asyncio.Semaphore(self.parallel_workers)
        
        async def process_batch(batch):
            async with semaphore:
                return await self.client.bulk_get_accounts(batch)
        
        tasks = [process_batch(batch) for batch in batches]
        batch_results = await asyncio.gather(*tasks, return_exceptions=True)
        
        # Combine and validate results
        for result in batch_results:
            if isinstance(result, Exception):
                logger.error(f"Batch processing failed: {result}")
            else:
                results.extend(result['data'])
        
        return results
```
