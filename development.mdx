## Development Environment Setup

This guide helps you set up a robust development environment for integrating with ADCB's Open Banking APIs.

<Info>
  **Prerequisites**: 
  - Programming language of choice (Python, Node.js, Java, etc.)
  - API testing tool (Postman, Insomnia, or cURL)
  - Valid ADCB developer account
  - mTLS certificates for your environment
</Info>

## Environment Configuration

### Sandbox vs Production

<Tabs>
  <Tab title="Sandbox Environment">
    **Base URL**: `https://sandbox-api.adcb.com`
    
    - Test with mock data
    - No real transactions
    - Relaxed rate limits
    - Self-signed certificates allowed
    - Test credentials provided
    
    ```bash
    # .env.sandbox
    ADCB_API_BASE_URL=https://sandbox-api.adcb.com
    ADCB_CLIENT_ID=sandbox_client_id
    ADCB_CLIENT_SECRET=sandbox_secret
    ADCB_CERT_PATH=./certs/sandbox-cert.pem
    ADCB_KEY_PATH=./certs/sandbox-key.pem
    ```
  </Tab>
  <Tab title="Production Environment">
    **Base URL**: `https://api.adcb.com`
    
    - Live customer data
    - Real transactions
    - Strict rate limits
    - CA-signed certificates required
    - Production credentials
    
    ```bash
    # .env.production
    ADCB_API_BASE_URL=https://api.adcb.com
    ADCB_CLIENT_ID=prod_client_id
    ADCB_CLIENT_SECRET=prod_secret
    ADCB_CERT_PATH=./certs/prod-cert.pem
    ADCB_KEY_PATH=./certs/prod-key.pem
    ```
  </Tab>
</Tabs>

## Quick Start Examples

### Initialize Client

<CodeGroup>

```python Python
from adcb import OpenBankingClient
import os

client = OpenBankingClient(
    client_id=os.environ['ADCB_CLIENT_ID'],
    client_secret=os.environ['ADCB_CLIENT_SECRET'],
    environment='sandbox',  # or 'production'
    cert_path=os.environ['ADCB_CERT_PATH'],
    key_path=os.environ['ADCB_KEY_PATH']
)

# Authenticate
token = client.authenticate(scopes=['accounts:read', 'transactions:read'])
```

```javascript Node.js
const { OpenBankingClient } = require('@adcb/openbanking-sdk');

const client = new OpenBankingClient({
    clientId: process.env.ADCB_CLIENT_ID,
    clientSecret: process.env.ADCB_CLIENT_SECRET,
    environment: 'sandbox', // or 'production'
    certPath: process.env.ADCB_CERT_PATH,
    keyPath: process.env.ADCB_KEY_PATH
});

// Authenticate
const token = await client.authenticate({
    scopes: ['accounts:read', 'transactions:read']
});
```

```java Java
import com.adcb.OpenBankingClient;

OpenBankingClient client = OpenBankingClient.builder()
    .clientId(System.getenv("ADCB_CLIENT_ID"))
    .clientSecret(System.getenv("ADCB_CLIENT_SECRET"))
    .environment(Environment.SANDBOX)
    .certPath(System.getenv("ADCB_CERT_PATH"))
    .keyPath(System.getenv("ADCB_KEY_PATH"))
    .build();

// Authenticate
Token token = client.authenticate(
    Arrays.asList("accounts:read", "transactions:read")
);
```

## Error Handling

Implement robust error handling for all API calls:

<CodeGroup>

```python Python
from adcb.exceptions import (
    AuthenticationError,
    RateLimitError,
    ValidationError,
    ServerError
)

try:
    accounts = client.accounts.list()
except AuthenticationError as e:
    # Token expired or invalid
    print(f"Auth error: {e.message}")
    # Refresh token and retry
    client.refresh_token()
except RateLimitError as e:
    # Too many requests
    print(f"Rate limit hit. Retry after: {e.retry_after}")
    time.sleep(e.retry_after)
except ValidationError as e:
    # Invalid request parameters
    print(f"Validation error: {e.errors}")
except ServerError as e:
    # ADCB server error
    print(f"Server error: {e.status_code}")
    # Implement exponential backoff
```

```javascript JavaScript
try {
    const accounts = await client.accounts.list();
} catch (error) {
    if (error.name === 'AuthenticationError') {
        // Token expired or invalid
        console.error('Auth error:', error.message);
        await client.refreshToken();
    } else if (error.name === 'RateLimitError') {
        // Too many requests
        console.error(`Rate limit hit. Retry after: ${error.retryAfter}`);
        await sleep(error.retryAfter * 1000);
    } else if (error.name === 'ValidationError') {
        // Invalid request parameters
        console.error('Validation error:', error.errors);
    } else if (error.name === 'ServerError') {
        // ADCB server error
        console.error('Server error:', error.statusCode);
        // Implement exponential backoff
    }
}
```

## Testing Strategies

### Unit Testing

<CodeGroup>

```python Python
import unittest
from unittest.mock import patch, Mock
from adcb import OpenBankingClient

class TestAccountOperations(unittest.TestCase):
    def setUp(self):
        self.client = OpenBankingClient(
            client_id="test_id",
            client_secret="test_secret",
            environment="sandbox"
        )
    
    @patch('adcb.accounts.list')
    def test_list_accounts(self, mock_list):
        # Mock response
        mock_list.return_value = [
            {"id": "ACC001", "balance": {"amount": 1000, "currency": "AED"}}
        ]
        
        accounts = self.client.accounts.list()
        
        self.assertEqual(len(accounts), 1)
        self.assertEqual(accounts[0]["id"], "ACC001")
```

```javascript Jest
const { OpenBankingClient } = require('@adcb/openbanking-sdk');

describe('Account Operations', () => {
    let client;
    
    beforeEach(() => {
        client = new OpenBankingClient({
            clientId: 'test_id',
            clientSecret: 'test_secret',
            environment: 'sandbox'
        });
    });
    
    test('should list accounts', async () => {
        // Mock the API call
        jest.spyOn(client.accounts, 'list').mockResolvedValue([
            { id: 'ACC001', balance: { amount: 1000, currency: 'AED' }}
        ]);
        
        const accounts = await client.accounts.list();
        
        expect(accounts).toHaveLength(1);
        expect(accounts[0].id).toBe('ACC001');
    });
});
```

## Logging and Monitoring

### Structured Logging

<CodeGroup>

```python Python
import logging
import json
from datetime import datetime

class ADCBLogger:
    def __init__(self):
        self.logger = logging.getLogger('adcb_api')
        handler = logging.StreamHandler()
        formatter = logging.Formatter(
            '%(asctime)s - %(name)s - %(levelname)s - %(message)s'
        )
        handler.setFormatter(formatter)
        self.logger.addHandler(handler)
        self.logger.setLevel(logging.INFO)
    
    def log_api_call(self, method, endpoint, status, duration):
        self.logger.info(json.dumps({
            'timestamp': datetime.utcnow().isoformat(),
            'method': method,
            'endpoint': endpoint,
            'status': status,
            'duration_ms': duration,
            'environment': os.environ.get('ENVIRONMENT', 'development')
        }))

logger = ADCBLogger()
```

```javascript Node.js
const winston = require('winston');

const logger = winston.createLogger({
    level: 'info',
    format: winston.format.combine(
        winston.format.timestamp(),
        winston.format.json()
    ),
    transports: [
        new winston.transports.Console(),
        new winston.transports.File({ filename: 'adcb-api.log' })
    ]
});

function logApiCall(method, endpoint, status, duration) {
    logger.info({
        method,
        endpoint,
        status,
        duration_ms: duration,
        environment: process.env.ENVIRONMENT || 'development'
    });
}
```

## Performance Optimization

### Connection Pooling

<CodeGroup>

```python Python
import requests
from requests.adapters import HTTPAdapter
from urllib3.util.retry import Retry

# Create session with connection pooling
session = requests.Session()

# Configure retries
retry_strategy = Retry(
    total=3,
    backoff_factor=1,
    status_forcelist=[429, 500, 502, 503, 504],
)

adapter = HTTPAdapter(
    max_retries=retry_strategy,
    pool_connections=10,
    pool_maxsize=20
)

session.mount("https://", adapter)

# Use session for all API calls
response = session.get(
    "https://api.adcb.com/v1/accounts",
    headers={"Authorization": f"Bearer {token}"}
)
```

```javascript Node.js
const https = require('https');
const axios = require('axios');

// Create axios instance with keep-alive
const client = axios.create({
    baseURL: 'https://api.adcb.com',
    timeout: 10000,
    httpsAgent: new https.Agent({
        keepAlive: true,
        maxSockets: 10,
        maxFreeSockets: 5,
        timeout: 60000,
        keepAliveTimeout: 30000
    })
});

// Add retry logic
axios.interceptors.response.use(null, async (error) => {
    if (error.config && error.response && error.response.status === 429) {
        const retryAfter = error.response.headers['retry-after'] || 1;
        await new Promise(resolve => setTimeout(resolve, retryAfter * 1000));
        return axios.request(error.config);
    }
    return Promise.reject(error);
});
```

### Caching Strategies

```python
from functools import lru_cache
from datetime import datetime, timedelta
import redis

class CacheManager:
    def __init__(self):
        self.redis_client = redis.Redis(
            host='localhost',
            port=6379,
            decode_responses=True
        )
    
    def cache_response(self, key, data, ttl=300):
        """Cache API response with TTL"""
        self.redis_client.setex(
            key,
            ttl,
            json.dumps(data)
        )
    
    def get_cached(self, key):
        """Retrieve cached response"""
        data = self.redis_client.get(key)
        return json.loads(data) if data else None
    
    @lru_cache(maxsize=128)
    def get_account_details(self, account_id):
        """LRU cache for frequently accessed accounts"""
        return self.client.accounts.get(account_id)
```
